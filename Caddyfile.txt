grizzlyrics.duckdns.org {
    root * /var/www/nextcloud
    php_fastcgi unix//run/php/php8.4-fpm.sock
    file_server
}

grizzly-stream.duckdns.org {

    # WoL script sur /wol
    handle_path /wol* {
        reverse_proxy 127.0.0.1:8080
    }

    # Jellyfin sur la racine
    handle_path /* {
        reverse_proxy 127.0.0.1:8096
    }

    log {
        output file /var/log/caddy/access.log
    }
}

beartify.duckdns.org {
    encode gzip

    header {
        X-Content-Type-Options "nosniff"
        X-Frame-Options "SAMEORIGIN"
    }

    # ── Proxy Jellyfin (clé injectée côté serveur) ──
    # IMPORTANT : ce bloc doit être AVANT le handle catch-all (file_server).
    # Les blocs handle sont mutuellement exclusifs dans Caddy v2 — une requête
    # qui matche /api/jellyfin/* ne passera jamais par file_server.
    handle /api/jellyfin/* {
        uri strip_prefix /api/jellyfin
        reverse_proxy 127.0.0.1:8096 {
            header_up X-Emby-Token "aaa8a7df4b364cf7bcc76f351d768798"
            header_up Host grizzly-stream.duckdns.org
        }
    }

    # ── Proxy Last.fm (clé injectée côté serveur) ──
    handle /api/lastfm/* {
        uri strip_prefix /api/lastfm
        rewrite * {path}?{query}&api_key=b25b959554ed76058ac220b7b2e0a026&format=json
        reverse_proxy https://ws.audioscrobbler.com {
            header_up Host ws.audioscrobbler.com
        }
    }

    # ── Proxy GrizzLyrics ──
    handle /api/lyrics/* {
        uri strip_prefix /api/lyrics
        reverse_proxy 127.0.0.1:80 {
            header_up Host grizzlyrics.duckdns.org
        }
    }

    # ── Fichiers statiques — catch-all (doit être EN DERNIER) ──
    # Encapsuler file_server dans handle {} l'empêche d'intercepter
    # les requêtes /api/* qui ont déjà été traitées ci-dessus.
    handle {
        root * /var/www/html/player
        file_server
    }
}
