# 1. On s'assure que les variables Android sont bien chargées
export ANDROID_HOME="$HOME/.android/sdk"
export NDK_HOME="$ANDROID_HOME/ndk/26.3.11579264"

# 2. On supprime le dossier Android corrompu/obsolète
rm -rf src-tauri/gen/android

# 3. On demande à Tauri de recréer proprement le projet Android
npx tauri android init

# 4. On lance la compilation finale de l'APK
npx tauri android build
