# 1. Créez l'arborescence complète attendue pour le manager Android
mkdir -p "$HOME/.android/sdk/cmdline-tools"

# 2. Rechargez l'environnement Android dans votre terminal actuel
export ANDROID_HOME="$HOME/.android/sdk"
export PATH="$ANDROID_HOME/cmdline-tools/latest/bin:$ANDROID_HOME/platform-tools:$PATH"

# 3. Relancez le script normalement
./build-linux.sh
