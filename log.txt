# 1. Le conteneur tourne et a fini de charger les modèles ?
docker ps -a | grep libretranslate
docker logs --tail 50 libretranslate

# 2. Test direct sur le conteneur (contourne Caddy)
curl -X POST http://127.0.0.1:3003/translate \
  -H "Content-Type: application/json" \
  -d '{"q":"hello world","source":"en","target":"fr","format":"text"}'

# 3. Test via Caddy en local (contourne le DNS/HTTPS public)
curl -X POST http://127.0.0.1/api/translate/translate \
  -H "Content-Type: application/json" \
  -H "Host: beartify.duckdns.org" \
  -d '{"q":"hello world","source":"en","target":"fr","format":"text"}'

# 4. Caddy a bien rechargé sans erreur ?
sudo systemctl status caddy --no-pager
sudo journalctl -u caddy --since "10 min ago" --no-pager | tail -30
